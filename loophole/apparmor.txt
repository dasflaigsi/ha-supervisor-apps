#include <tunables/global>

profile ha-loophole-addon flags=(attach_disconnected,mediate_deleted) {

  # HA/S6 infrastructure
  #include <abstractions/base>

  signal,

  /init rix,
  /bin/** rix,
  /usr/bin/** rix,
  /run/{s6,s6-rc*,service}/** rwix,
  /package/** rix,
  /command/** rix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/** rwixk,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** rix,

  # Application
  /usr/src/app/ rix,
  /usr/src/app/** rix,
  /data/** rw,
  /proc/ r,
  /proc/** r,

  # Basic runtime configuration
  /etc/** r,
  /usr/share/ca-certificates/** r,
  /usr/share/zoneinfo/** r,

  # Networking
  network inet stream,
  network inet dgram,
  network inet6 stream,
  network inet6 dgram,

  # Loophole
  /usr/local/bin/loophole cx -> loophole,

  profile loophole flags=(attach_disconnected,mediate_deleted) {

    # Allow the CLI to execute itself while remaining in this profile.
    /usr/local/bin/loophole rix,

    # Loophole's persistent state
    /data/loophole-home/** rw,

    # Loophole log
    /data/loophole.log rw,

    # Runtime / TLS
    /etc/** r,
    /usr/share/ca-certificates/** r,
    /usr/share/zoneinfo/** r,
    /lib/** mr,
    /lib64/** mr,
    /usr/lib/** mr,

    # Required by the musl ELF interpreter
    /lib/ld-musl-x86_64.so.1 rix,

    # Network
    network inet stream,
    network inet6 stream,

    # Parent may stop/restart it
    signal (receive),
  }
}
